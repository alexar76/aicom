+def esim_sellers_from_search(payload: Any) -> dict[str, dict[str, Any]]:
+    """{normalised seller url: what our hub knows about its eSIM listing} from a hub search.
+
+    Only `esim.quote@*` rows count, and only the hub's own fields are read: the seller's
+    name as the hub crawled it, its product id, trust, and the deliveries buyers confirmed
+    with signed receipts (`deliveries_30d`). Nothing the seller merely claims.
+    """
+    out: dict[str, dict[str, Any]] = {}
+    matches = payload.get("matches") if isinstance(payload, dict) else None
+    if not isinstance(matches, list):
+        return out
+    for row in matches:
+        if not isinstance(row, dict):
+            continue
+        if not str(row.get("capability_id") or "").startswith(_ESIM_QUOTE_PREFIX):
+            continue
+        key = _norm_url(row.get("source_hub"))
+        if not key or key in out:
+            continue
+        raw = row.get("deliveries_30d") if isinstance(row.get("deliveries_30d"), dict) else {}
+        deliveries = {k: _count(raw.get(k)) or 0 for k in ("confirmed", "distinct_buyers", "refunded")}
+        out[key] = {
+            "product_id": str(row.get("product_id") or "")[:80],
+            "name": str(row.get("source_hub_name") or "")[:80],
+            "trust_score": _num(row.get("trust_score")),
+            "deliveries_30d": deliveries,
+        }
+    return out
